-- Create Table 
CREATE TABLE Restaurant (
    Restaurant_Name VARCHAR2(30) PRIMARY KEY,
    Restaurant_Location VARCHAR2(100),
    Telephone CHAR(11)
);

CREATE TABLE DeliveryMan (
    DeliveryMan_ID CHAR(10) PRIMARY KEY,
    DeliveryMan_Name VARCHAR2(50),
    DeliveryMan_Phone CHAR(11),
    Vehicle_No VARCHAR2(15),
    Restaurant_Name VARCHAR2(30),
    FOREIGN KEY (Restaurant_Name) REFERENCES Restaurant(Restaurant_Name)
);

CREATE TABLE Customer (
    Customer_ID CHAR(10) PRIMARY KEY,
    Customer_Name VARCHAR2(50),
    Email_Address VARCHAR2(63),
    Contact_No CHAR(11),
    Customer_Address VARCHAR2(100),
    DeliveryMan_ID CHAR(10),
    FOREIGN KEY (DeliveryMan_ID) REFERENCES DeliveryMan(DeliveryMan_ID)
);

CREATE TABLE Browse (
    Customer_ID CHAR(10),
    Restaurant_Name VARCHAR2(30),
    PRIMARY KEY (Customer_ID, Restaurant_Name),
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID),
    FOREIGN KEY (Restaurant_Name) REFERENCES Restaurant(Restaurant_Name)
);

CREATE TABLE Kitchen (
    Kitchen_ID CHAR(10) PRIMARY KEY,
    Kitchen_Status VARCHAR2(20)
);

CREATE TABLE FoodItem (
    Food_ID CHAR(3) PRIMARY KEY,
    Food_Name VARCHAR2(30),
    Price NUMBER(10),
    Category VARCHAR2(11),
    Restaurant_Name VARCHAR2(30),
    Kitchen_ID CHAR(10),
    FOREIGN KEY (Restaurant_Name) REFERENCES Restaurant(Restaurant_Name),
    FOREIGN KEY (Kitchen_ID) REFERENCES Kitchen(Kitchen_ID)
);

CREATE TABLE Purchase (
    Purchase_ID CHAR(10) PRIMARY KEY,
    Purchase_Date DATE,
    Amount NUMBER(10),
    Customer_ID CHAR(10),
    DeliveryMan_ID CHAR(10),
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID),
    FOREIGN KEY (DeliveryMan_ID) REFERENCES DeliveryMan(DeliveryMan_ID)
);

CREATE TABLE Contains (
    Purchase_ID CHAR(10),
    Food_ID CHAR(3),
    PRIMARY KEY (Purchase_ID, Food_ID),
    FOREIGN KEY (Purchase_ID) REFERENCES Purchase(Purchase_ID),
    FOREIGN KEY (Food_ID) REFERENCES FoodItem(Food_ID)
);

-- Query 01: Show customers with delivery men from restaurants located in Dhaka and selected restaurants.
SELECT C.Customer_Name,D.DeliveryMan_Name,R.Restaurant_Name
FROM Customer C
JOIN DeliveryMan D ON C.DeliveryMan_ID=D.DeliveryMan_ID
JOIN Restaurant R ON D.Restaurant_Name=R.Restaurant_Name
WHERE R.Restaurant_Location LIKE 'Dhaka%'
AND R.Restaurant_Name IN ('PizzaHut','KFC','Dominos');

-- Query 02: Show Mirpur or Gulshan customers whose purchase is above average
SELECT C.Customer_Name,P.Amount
FROM Customer C
JOIN Purchase P ON C.Customer_ID=P.Customer_ID
WHERE (C.Customer_Address='Dhaka Mirpur' OR C.Customer_Address='Dhaka Gulshan')
AND P.Amount>(SELECT AVG(Amount) FROM Purchase);

-- Query 03: Show customers who browsed PizzaHut and made an above-average purchase.
SELECT C.Customer_Name,B.Restaurant_Name,P.Amount
FROM Customer C
JOIN Browse B ON C.Customer_ID=B.Customer_ID
JOIN Purchase P ON C.Customer_ID=P.Customer_ID
WHERE B.Restaurant_Name='PizzaHut'
AND P.Amount>(SELECT AVG(Amount) FROM Purchase);

-- Query 04: Show customers whose delivery restaurants are not KFC or BurgerKing and whose purchase exceeds 200.
SELECT C.Customer_Name,D.DeliveryMan_Name,P.Amount
FROM Customer C
JOIN DeliveryMan D ON C.DeliveryMan_ID=D.DeliveryMan_ID
JOIN Purchase P ON C.Customer_ID=P.Customer_ID
WHERE D.Restaurant_Name NOT IN ('KFC','BurgerKing')
AND P.Amount>200;

-- Query 05: Show non-closed Italian or FastFood items above average price with restaurant and kitchen information.
SELECT F.Food_Name,F.Price,R.Restaurant_Name,K.Kitchen_Status
FROM FoodItem F
JOIN Restaurant R ON F.Restaurant_Name=R.Restaurant_Name
JOIN Kitchen K ON F.Kitchen_ID=K.Kitchen_ID
WHERE (F.Category='Italian' OR F.Category='FastFood')
AND F.Price>(SELECT AVG(Price) FROM FoodItem)
AND K.Kitchen_Status<>'Closed';

-- Query 06: Show customers who purchased food above 200 from KFC, PizzaHut or Dominos.
SELECT DISTINCT C.Customer_Name,R.Restaurant_Name,F.Food_Name,P.Amount
FROM Customer C
JOIN Purchase P ON C.Customer_ID=P.Customer_ID
JOIN Contains CO ON P.Purchase_ID=CO.Purchase_ID

JOIN Restaurant R ON F.Restaurant_Name=R.Restaurant_Name
WHERE F.Price>200
AND R.Restaurant_Name IN ('KFC','PizzaHut','Dominos');

-- Query 22: Show customers whose total spending is greater than the average customer spending.
SELECT C.Customer_Name,SUM(P.Amount) AS Total_Amount
FROM Customer C
JOIN Purchase P ON C.Customer_ID=P.Customer_ID
GROUP BY C.Customer_Name
HAVING SUM(P.Amount)>(
    SELECT AVG(Total_Amount)
    FROM(
        SELECT SUM(Amount) AS Total_Amount
        FROM Purchase
        GROUP BY Customer_ID
    )
);

-- Query 48: Show restaurants with browsing customers, more than one food item and average food price above 200.
SELECT R.Restaurant_Name,COUNT(DISTINCT B.Customer_ID) AS Browsed_Customers,COUNT(DISTINCT F.Food_ID) AS Food_Count,AVG(F.Price) AS Average_Price
FROM Restaurant R
LEFT JOIN Browse B ON R.Restaurant_Name=B.Restaurant_Name

HAVING COUNT(DISTINCT B.Customer_ID)>0

-- Query 49: Show customers, delivery men, restaurants and food items where both purchase and food price are above average and customer browsed.

-- Query 50: Write a PL/SQL block to calculate and display the total sales amount and average purchase amount from the Purchase table.

SET SERVEROUTPUT ON;

DECLARE
    total_sales NUMBER;
    avg_purchase NUMBER;
BEGIN
    SELECT SUM(Amount),AVG(Amount)
    INTO total_sales,avg_purchase
    FROM Purchase;

    DBMS_OUTPUT.PUT_LINE('Total Sales: '||total_sales);
    DBMS_OUTPUT.PUT_LINE('Average Purchase: '||avg_purchase);
END;

-- Query 51: Write a PL/SQL block to find and display the highest purchase amount along with the customer name who made that purchase.
DECLARE
    highest_amount NUMBER;
    customer_name VARCHAR2(50);
BEGIN
    SELECT MAX(Amount)
    INTO highest_amount
    FROM Purchase;

    SELECT Customer_Name
    INTO customer_name
    FROM Customer
    WHERE Customer_ID=(
        SELECT Customer_ID
        FROM Purchase
        WHERE Amount=highest_amount
        AND ROWNUM=1
    );

    DBMS_OUTPUT.PUT_LINE('Highest Purchase: '||highest_amount);
    DBMS_OUTPUT.PUT_LINE('Customer: '||customer_name);
END;

-- Query 52: Write a PL/SQL block using a FOR loop cursor to display customers whose total purchase amount is greater than 250, sorted by total amount in descending order.
BEGIN
    FOR X IN(
        SELECT C.Customer_Name,SUM(P.Amount) AS Total_Amount
        FROM Customer C
        JOIN Purchase P ON C.Customer_ID=P.Customer_ID
        GROUP BY C.Customer_Name
        HAVING SUM(P.Amount)>250
        ORDER BY Total_Amount DESC
    )
    LOOP
        DBMS_OUTPUT.PUT_LINE(X.Customer_Name||' : '||X.Total_Amount);
    END LOOP;
END;
